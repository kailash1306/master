pipeline{
    agent any
    stages{
        stage('checkout'){
            steps{
                git branch: 'main', url: 'https://github.com/kailash1306/master.git'
            }
        }
        stage('Install dependencies'){
            steps{
                bat 'pip install -r requirements.txt'
            }
        }
        stage('Run Unit Tests'){
            steps{
                bat 'pytest test_app.py'
            }
        }
    }
